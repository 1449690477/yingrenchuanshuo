## 绿幕规范化命令

内置 ImageGen 返回方图后，统一缩到 1024 × 1024，并把检测到的纯背景区规范为精确 `#00ff00`。该步骤只处理五张 `*-chroma.png` 母图。

```powershell
@'
from pathlib import Path
from PIL import Image
import numpy as np

names = [
    "sigil_swordsman",
    "sigil_witch",
    "sigil_shaman",
    "sigil_catkin",
    "crystal_resonance",
]
base = Path("art-source/reforge")

for name in names:
    path = base / f"{name}-chroma.png"
    with Image.open(path) as im:
        rgb = im.convert("RGB").resize((1024, 1024), Image.Resampling.LANCZOS)
    arr = np.array(rgb)
    border = np.concatenate((
        arr[:12].reshape(-1, 3),
        arr[-12:].reshape(-1, 3),
        arr[:, :12].reshape(-1, 3),
        arr[:, -12:].reshape(-1, 3),
    ))
    key = np.median(border, axis=0)
    dist = np.linalg.norm(arr.astype(np.float32) - key.astype(np.float32), axis=2)
    strong_green = (
        (arr[:, :, 1] > 180)
        & (arr[:, :, 1] > arr[:, :, 0] * 3)
        & (arr[:, :, 1] > arr[:, :, 2] * 3)
    )
    arr[(dist <= 48) & strong_green] = (0, 255, 0)
    Image.fromarray(arr, "RGB").save(path, format="PNG", optimize=True, compress_level=9)
'@ | python -
```

## 256 × 256 RGBA 压缩命令

使用预乘 Alpha 缩放，避免透明边缘混入绿幕颜色；全透明像素的 RGB 清零以提高 PNG 压缩率。

```powershell
@'
from pathlib import Path
from PIL import Image
import numpy as np

names = [
    "sigil_swordsman",
    "sigil_witch",
    "sigil_shaman",
    "sigil_catkin",
    "crystal_resonance",
]
src_dir = Path(r"C:\Users\Administrator\AppData\Local\Temp\sakura-reforge-sigils-a")
out_dir = Path("public/assets/items")

for name in names:
    with Image.open(src_dir / f"{name}-alpha.png") as im:
        rgba = im.convert("RGBA")
        small = (
            rgba.convert("RGBa")
            .resize((256, 256), Image.Resampling.LANCZOS)
            .convert("RGBA")
        )
    arr = np.array(small)
    arr[arr[:, :, 3] == 0, :3] = 0
    final = Image.fromarray(arr, "RGBA")
    final.save(
        out_dir / f"{name}.png",
        format="PNG",
        optimize=True,
        compress_level=9,
    )
'@ | python -
```
